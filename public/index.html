① カードデータを読み込む

  "settings": { "mainDeckSize": 40, "openingHand": 5, "startingLife": 8 },
  "cards": [
    { "id": "chr-01", "type": "character", "name": "名前", "text": "効果テキスト", "image": "https://... または /uploads/... (任意)" }
  ]
}
